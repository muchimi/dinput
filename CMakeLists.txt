cmake_minimum_required(VERSION 3.15)
project(DILL LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(dill SHARED src/dill.cpp dill.rc)
target_compile_definitions(dill PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(dill PRIVATE dinput8 dxguid ole32)
if(MSVC)
  target_compile_options(dill PRIVATE /W3 /Zi)
  target_link_options(dill PRIVATE /DEBUG)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(dill_hotplug_test tests/hotplug_test.cpp)
  target_compile_definitions(dill_hotplug_test PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(dill_hotplug_test PRIVATE dinput8 dxguid ole32)
  add_test(NAME dill_hotplug COMMAND dill_hotplug_test)
endif()

option(DILL_BUILD_EXAMPLES "Build interactive device examples" OFF)
if(DILL_BUILD_EXAMPLES)
  add_executable(example src/example.cpp)
  target_link_libraries(example PRIVATE dill)
  add_executable(example2 src/example2.cpp)
  target_link_libraries(example2 PRIVATE dill)
endif()
